// ─── Generation Error Category ───────────────────────────────────────────────

  "quota",
  "schema",
  "build",
  "rate_limit",
  "network",
  "internal",
] as const;
 
export type GenerationErrorCategory = (typeof GENERATION_ERROR_CATEGORIES)[number];
 
// ─── Generation Status ───────────────────────────────────────────────────────
export type GenerationStatus =
  | "pending"
  | "extracting_schema"
  | "generating_code"
  | "generating" // build-retry alias of generating_code (worker re-calls the LLM)
  | "building"
  | "packing"
  | "uploading"
  | "success"
  | "failed";

export interface Generation {
  id: string;

  /** Tier 0 (Spark/free) only: generated file contents keyed by relative path.

  /** Personalization choices stored at generation time */
  personalization_json: PersonalizationData | null;

  build_log: string | null;
  error_message: string | null;
  error_category: GenerationErrorCategory | null;
  attempt_count: number;
  created_at: string;
  updated_at: string;
  completed_at: string | null;
}
