using Dapper;
using InvoiceHub.Infrastructure;
using InvoiceHub.Models;

namespace InvoiceHub.Repositories;

public interface ICustomerRepository
{
    Task<IEnumerable<Customer>> GetAllAsync();
    Task<Customer?> GetByIdAsync(Guid id);
    Task<Customer> CreateAsync(CreateCustomerRequest request);
    Task<bool> UpdateAsync(Guid id, CreateCustomerRequest request);
    Task<bool> DeleteAsync(Guid id);
}

public sealed class CustomerRepository(IDbConnectionFactory db) : ICustomerRepository
{
    public async Task<IEnumerable<Customer>> GetAllAsync()
    {
        using var conn = db.CreateConnection();
        return await conn.QueryAsync<Customer>(
            "SELECT id, name, email, created_at AS CreatedAt FROM customers ORDER BY created_at DESC");
    }

    public async Task<Customer?> GetByIdAsync(Guid id)
    {
        using var conn = db.CreateConnection();
        return await conn.QueryFirstOrDefaultAsync<Customer>(
            "SELECT id, name, email, created_at AS CreatedAt FROM customers WHERE id = @Id",
            new { Id = id });
    }

    public async Task<Customer> CreateAsync(CreateCustomerRequest request)
    {
        using var conn = db.CreateConnection();
        var id = Guid.NewGuid();
        await conn.ExecuteAsync(
            "INSERT INTO customers (id, name, email, created_at) VALUES (@Id, @Name, @Email, @CreatedAt)",
            new { Id = id, request.Name, request.Email, CreatedAt = DateTime.UtcNow });
        return (await GetByIdAsync(id))!;
    }

    public async Task<bool> UpdateAsync(Guid id, CreateCustomerRequest request)
    {
        using var conn = db.CreateConnection();
        var rows = await conn.ExecuteAsync(
            "UPDATE customers SET name = @Name, email = @Email WHERE id = @Id",
            new { Id = id, request.Name, request.Email });
        return rows > 0;
    }

    public async Task<bool> DeleteAsync(Guid id)
    {
        using var conn = db.CreateConnection();
        var rows = await conn.ExecuteAsync("DELETE FROM customers WHERE id = @Id", new { Id = id });
        return rows > 0;
    }
}

using InvoiceHub.Models;
using InvoiceHub.Repositories;

namespace InvoiceHub.Controllers;

public static class CustomerEndpoints
{
    public static void MapCustomerEndpoints(this WebApplication app)
    {
        var group = app.MapGroup("/api/v1/customers").WithTags("Customer");

        group.MapGet("/", async (ICustomerRepository repo) =>
            Results.Ok(await repo.GetAllAsync()));

        group.MapGet("/{id:guid}", async (Guid id, ICustomerRepository repo) =>
            await repo.GetByIdAsync(id) is { } customer ? Results.Ok(customer) : Results.NotFound());

        group.MapPost("/", async (CreateCustomerRequest request, ICustomerRepository repo) =>
        {
            var created = await repo.CreateAsync(request);
            return Results.Created($"/api/v1/customers/{created.Id}", created);
        });

        group.MapPut("/{id:guid}", async (Guid id, CreateCustomerRequest request, ICustomerRepository repo) =>
            await repo.UpdateAsync(id, request) ? Results.NoContent() : Results.NotFound());

        group.MapDelete("/{id:guid}", async (Guid id, ICustomerRepository repo) =>
            await repo.DeleteAsync(id) ? Results.NoContent() : Results.NotFound());
    }
}

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TABLE customers (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name TEXT NOT NULL,
    email TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE invoices (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    customer_id UUID NOT NULL,
    number TEXT NOT NULL,
    total NUMERIC(10,2) NOT NULL,
    issued_at TIMESTAMPTZ NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE lineitems (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    invoice_id UUID NOT NULL,
    description TEXT NOT NULL,
    quantity INTEGER NOT NULL,
    unit_price NUMERIC(10,2) NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

ALTER TABLE invoices ADD CONSTRAINT fk_invoices_customers
    FOREIGN KEY (customer_id) REFERENCES customers(id) ON DELETE CASCADE;

ALTER TABLE lineitems ADD CONSTRAINT fk_lineitems_invoices
    FOREIGN KEY (invoice_id) REFERENCES invoices(id) ON DELETE CASCADE;

ALTER TABLE customers ENABLE ROW LEVEL SECURITY;
ALTER TABLE invoices ENABLE ROW LEVEL SECURITY;
ALTER TABLE lineitems ENABLE ROW LEVEL SECURITY;

import type {
  CreateCustomerInput,
  CreateInvoiceInput,
  CreateLineItemInput,
  Customer,
  Invoice,
  LineItem,
} from "@/types";

const API_URL = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:5000";

export async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${API_URL}${path}`, {
    headers: { "Content-Type": "application/json", ...init?.headers },
    ...init,
  });
  if (!res.ok) throw new Error(`API error ${res.status}: ${await res.text()}`);
  return res.json() as Promise<T>;
}

export async function getCustomers(): Promise<Customer[]> {
  return apiFetch<Customer[]>("/api/v1/customers");
}

  return apiFetch<Customer>(`/api/v1/customers/${id}`);
}

  return apiFetch<Customer>("/api/v1/customers", {
